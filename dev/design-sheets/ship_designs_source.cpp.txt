#include "ships.h"

// Candidate designs. Top-down, symmetrical, +x is forward. Each hull is given as
// the upper half of its outline from the nose (y = 0) to the tail (y = 0); the
// lower half is mirrored automatically.
static void sym(SDL_Surface *s, int x, int y, int u, int d, const ShipPoint *h, int n,
	Uint8 r, Uint8 g, Uint8 b, Uint8 a, Uint8 e) {
	ShipPoint p[24];
	int c = 0;
	for(int i = 0; i < n; i++) { p[c][0] = h[i][0]; p[c][1] = h[i][1]; c++; }
	for(int i = n - 2; i >= 1; i--) { p[c][0] = h[i][0]; p[c][1] = (signed char)-h[i][1]; c++; }
	ship_shape(s, x, y, u, d, p, c, r, g, b, a, e);
}

#define DESIGN(name) static void name(SDL_Surface *s, int x, int y, int u, int d, \
	Uint8 r, Uint8 g, Uint8 b, Uint8 a)
#define SPINE(h, n) sym(s, x, y, u, d, h, n, ship_light(r, 12), ship_light(g, 12), \
	ship_light(b, 12), a, a / 3)
#define RED 196, 56, 42

DESIGN(dagger) {
	ship_set_palette(&r, &g, &b);
	static const ShipPoint hull[] = { {45,0},{30,-3},{12,-5},{-4,-14},{-20,-22},{-14,-8},{-26,-5},{-28,0} };
	static const ShipPoint spine[] = { {40,0},{18,-2},{-20,-2},{-28,0} };
	sym(s, x, y, u, d, hull, 8, r, g, b, a, a / 2);
	SPINE(spine, 4);
	ship_line(s, x, y, u, d, 34, 0, 22, 0, RED, (Uint8)(a * 0.9));
	ship_engine(s, x, y, u, d, -28, -3, 2, a);
	ship_engine(s, x, y, u, d, -28, 3, 2, a);
}

DESIGN(leviathan) {
	ship_set_palette(&r, &g, &b);
	static const ShipPoint hull[] = { {52,0},{38,-4},{12,-7},{-30,-8},{-46,-5},{-50,0} };
	static const ShipPoint spine[] = { {48,0},{12,-3},{-46,-2},{-50,0} };
	int spike_x[4] = { 20, 4, -12, -28 };
	int spike_h[4] = { 14, 18, 18, 14 };
	for(int i = 0; i < 4; i++) {
		for(int side = -1; side <= 1; side += 2) {
			ShipPoint p[3] = { {(signed char)spike_x[i], (signed char)(side * 7)},
				{(signed char)(spike_x[i] - 5), (signed char)(side * spike_h[i])},
				{(signed char)(spike_x[i] - 10), (signed char)(side * 7)} };
			ship_shape(s, x, y, u, d, p, 3, r, g, b, a, a / 2);
		}
	}
	sym(s, x, y, u, d, hull, 6, r, g, b, a, a / 2);
	SPINE(spine, 4);
	ship_line(s, x, y, u, d, 44, 0, 30, 0, RED, (Uint8)(a * 0.9));
	ship_engine(s, x, y, u, d, -50, -2, 2, a);
	ship_engine(s, x, y, u, d, -50, 2, 2, a);
}

typedef void (*Design)(SDL_Surface *, int, int, int, int, Uint8, Uint8, Uint8, Uint8);

int main() {
	static const struct { const char *name; Design draw; } designs[10] = {
		{ "1  Dagger", dagger }, { "2  Manta", manta }, { "3  Hammerhead", hammerhead },
		{ "4  Mandible", mandible }, { "5  Trident", trident }, { "6  Scythe", scythe },
		{ "7  Bat", bat }, { "8  Leviathan", leviathan }, { "9  Wedge", wedge },
		{ "10  Forward-swept", forward_swept } };
	const int cols = 5, cell_w = 400, cell_h = 330;
	int W = cols * cell_w, H = 2 * cell_h;
	SDL_Surface *s = SDL_CreateRGBSurface(SDL_SWSURFACE, W, H, 32,
		0x00ff0000, 0x0000ff00, 0x000000ff, 0xff000000);
	for(int yy = 0; yy < H; yy++) {
		int row_y = yy % cell_h;
		SDL_Rect r = { 0, (Sint16)yy, (Uint16)W, 1 };
		SDL_FillRect(s, &r, SDL_MapRGB(s->format, 16 + row_y / 22, 22 + row_y / 18, 46 + row_y / 12));
	}
	unsigned int seed = 12345;
	for(int i = 0; i < 260; i++) {
		seed = seed * 1664525u + 1013904223u;
		int sx = seed % W;
		seed = seed * 1664525u + 1013904223u;
		pixelRGBA(s, sx, seed % H, 205, 214, 235, 90);
	}
	for(int i = 0; i < 10; i++) {
		int cx = (i % cols) * cell_w, cy = (i / cols) * cell_h;
		vlineRGBA(s, cx, cy, cy + cell_h - 1, 70, 80, 110, 160);
		hlineRGBA(s, cx, cx + cell_w - 1, cy, 70, 80, 110, 160);
		stringRGBA(s, cx + 12, cy + 10, designs[i].name, 235, 235, 245, 255);
		designs[i].draw(s, cx + cell_w / 2, cy + 125, 24, 1, 100, 147, 167, 230);
		stringRGBA(s, cx + 12, cy + 245, "in game size:", 150, 160, 185, 255);
		designs[i].draw(s, cx + 110, cy + 295, 10, 1, 100, 147, 167, 150);
		designs[i].draw(s, cx + 290, cy + 295, 10, -1, 170, 126, 101, 150);
	}
	SDL_SaveBMP(s, "designs.bmp");
	return 0;
}
